/**

import { useState, useEffect, useRef, useCallback } from 'react';
import { SIMULATION_TOPICS, getTopicById } from './topics/registry';
import {
  SimulationTopic,
  ExecutionMode,
  SimulationStatus,
  ExplanationDepth,
  SimulationSnapshot,
  ValidationResult,
  ScenarioPreset,
} from './types/simulation';
import { Header, WorkbenchView } from './components/Header';
import { ControlDeck } from './components/ControlDeck';
import { Workbench } from './components/Workbench';
import { ParameterPanel } from './components/ParameterPanel';
import { VisualizerRouter } from './components/VisualizerRouter';
import { TelemetryMatrix } from './components/TelemetryMatrix';
import { TimelineEventLog } from './components/TimelineEventLog';
import { DeepInspector } from './components/DeepInspector';
import { ComparisonModal } from './components/ComparisonModal';
import { TopicDocsModal } from './components/TopicDocsModal';

export default function App() {
  // Topic state
  const [activeTopic, setActiveTopic] = useState<SimulationTopic>(SIMULATION_TOPICS[0]);
  const [activePreset, setActivePreset] = useState<ScenarioPreset>(activeTopic.presets[0]);

  // Code state
  const [benchmarkCode, setBenchmarkCode] = useState<string>(activeTopic.defaultCode);
  const [editCode, setEditCode] = useState<string>(activeTopic.defaultCode);
  const [validationResult, setValidationResult] = useState<ValidationResult>({ isValid: true, issues: [] });

  // Parameters state
  const [paramValues, setParamValues] = useState<Record<string, any>>(() => {
    const initial: Record<string, any> = {};
    activeTopic.parameters.forEach((p) => {
      initial[p.id] = p.defaultValue;
    });
    return initial;
  });

  // UI state
  const [activeView, setActiveView] = useState<WorkbenchView>('split');
  const [explanationDepth, setExplanationDepth] = useState<ExplanationDepth>('technical');
  const [isParamsOpen, setIsParamsOpen] = useState(false);
  const [isDocsOpen, setIsDocsOpen] = useState(false);
  const [isComparisonOpen, setIsComparisonOpen] = useState(false);

  // Simulation execution state
  const [mode, setMode] = useState<ExecutionMode>('DEFAULT');
  const [status, setStatus] = useState<SimulationStatus>('idle');
  const [currentStep, setCurrentStep] = useState<number>(0);
  const [playbackSpeed, setPlaybackSpeed] = useState<number>(1);

  // Simulation plan snapshots
  const [snapshots, setSnapshots] = useState<SimulationSnapshot[]>(() => {
    const { steps } = activeTopic.buildExecutionPlan(activeTopic.defaultCode, {});
    return steps;
  });

  // Baseline benchmark snapshots for comparison
  const [benchmarkSnapshots, setBenchmarkSnapshots] = useState<SimulationSnapshot[]>(snapshots);

  // Initialize or change topic
  const handleSelectTopic = (topicId: string) => {
    const newTopic = getTopicById(topicId);
    setActiveTopic(newTopic);
    const firstPreset = newTopic.presets[0];
    setActivePreset(firstPreset);
    setBenchmarkCode(firstPreset ? firstPreset.code : newTopic.defaultCode);
    setEditCode(firstPreset ? firstPreset.code : newTopic.defaultCode);

    const initialParams: Record<string, any> = {};
    newTopic.parameters.forEach((p) => {
      initialParams[p.id] = firstPreset?.parameters?.[p.id] ?? p.defaultValue;
    });
    setParamValues(initialParams);

    const { steps } = newTopic.buildExecutionPlan(firstPreset ? firstPreset.code : newTopic.defaultCode, initialParams);
    setSnapshots(steps);
    setBenchmarkSnapshots(steps);
    setCurrentStep(0);
    setMode('DEFAULT');
    setStatus('idle');
  };

  // Change preset scenario
  const handleSelectPreset = (preset: ScenarioPreset) => {
    setActivePreset(preset);
    setBenchmarkCode(preset.code);
    setEditCode(preset.code);

    const newParams = { ...paramValues, ...(preset.parameters || {}) };
    setParamValues(newParams);

    const { steps } = activeTopic.buildExecutionPlan(preset.code, newParams);
    setSnapshots(steps);
    setBenchmarkSnapshots(steps);
    setCurrentStep(0);
    setMode('DEFAULT');
    setStatus('idle');
  };

  // Parameter changes
  const handleParamChange = (id: string, value: any) => {
    const updated = { ...paramValues, [id]: value };
    setParamValues(updated);

    // Rebuild plan with updated parameters
    const currentCode = mode === 'CUSTOM' ? editCode : benchmarkCode;
    const { steps } = activeTopic.buildExecutionPlan(currentCode, updated);
    setSnapshots(steps);
    if (mode === 'DEFAULT') {
      setBenchmarkSnapshots(steps);
    }
  };

  const handleResetParams = () => {
    const initialParams: Record<string, any> = {};
    activeTopic.parameters.forEach((p) => {
      initialParams[p.id] = activePreset?.parameters?.[p.id] ?? p.defaultValue;
    });
    setParamValues(initialParams);

    const currentCode = mode === 'CUSTOM' ? editCode : benchmarkCode;
    const { steps } = activeTopic.buildExecutionPlan(currentCode, initialParams);
    setSnapshots(steps);
  };

  // Execute custom input manually (#9)
  const handleExecuteCustom = () => {
    const valResult = activeTopic.validate(editCode, paramValues);
    setValidationResult(valResult);

    if (!valResult.isValid) {
      setStatus('error');
      return;
    }

    const { steps } = activeTopic.buildExecutionPlan(editCode, paramValues);
    setSnapshots(steps);
    setMode('CUSTOM');
    setCurrentStep(0);
    setStatus('paused');
  };

  // Copy benchmark to edit space (#6)
  const handleCopyToEditSpace = () => {
    setEditCode(benchmarkCode);
  };

  // Reset to default canonical benchmark (#13)
  const handleResetToDefault = () => {
    const code = activePreset ? activePreset.code : activeTopic.defaultCode;
    setEditCode(code);
    setBenchmarkCode(code);
    setValidationResult({ isValid: true, issues: [] });

    const initialParams: Record<string, any> = {};
    activeTopic.parameters.forEach((p) => {
      initialParams[p.id] = activePreset?.parameters?.[p.id] ?? p.defaultValue;
    });
    setParamValues(initialParams);

    const { steps } = activeTopic.buildExecutionPlan(code, initialParams);
    setSnapshots(steps);
    setBenchmarkSnapshots(steps);
    setMode('DEFAULT');
    setCurrentStep(0);
    setStatus('idle');
  };

  // Step navigation (#11, #12)
  const handleNextStep = useCallback(() => {
    setCurrentStep((prev) => {
      if (prev < snapshots.length - 1) {
        return prev + 1;
      }
      setStatus('completed');
      return prev;
    });
  }, [snapshots.length]);

  const handlePrevStep = useCallback(() => {
    setCurrentStep((prev) => {
      if (prev > 0) {
        if (status === 'completed') setStatus('paused');
        return prev - 1;
      }
      return 0;
    });
  }, [status]);

  const handleReset = () => {
    setCurrentStep(0);
    setStatus('paused');
  };

  const handlePlay = () => {
    if (currentStep >= snapshots.length - 1) {
      setCurrentStep(0);
    }
    setStatus('running');
  };

  const handlePause = () => {
    setStatus('paused');
  };

  // Playback timer loop
  useEffect(() => {
    let timer: any;
    if (status === 'running') {
      const intervalMs = Math.round(1800 / playbackSpeed);
      timer = setInterval(() => {
        setCurrentStep((prev) => {
          if (prev < snapshots.length - 1) {
            return prev + 1;
          } else {
            setStatus('completed');
            return prev;
          }
        });
      }, intervalMs);
    }
    return () => clearInterval(timer);
  }, [status, playbackSpeed, snapshots.length]);

  // Global Keyboard Controls (#56)
  useEffect(() => {
    const handleKeyDown = (e: KeyboardEvent) => {
      // Don't intercept if user is typing in the textarea
      if (
        document.activeElement?.tagName === 'TEXTAREA' ||
        document.activeElement?.tagName === 'INPUT'
      ) {
        return;
      }

      if (e.code === 'Space') {
        e.preventDefault();
        if (status === 'running') {
          handlePause();
        } else {
          handlePlay();
        }
      } else if (e.key === 'ArrowRight') {
        e.preventDefault();
        handleNextStep();
      } else if (e.key === 'ArrowLeft') {
        e.preventDefault();
        handlePrevStep();
      } else if (e.key.toLowerCase() === 'r') {
        e.preventDefault();
        handleReset();
      }
    };

    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [status, handleNextStep, handlePrevStep]);

  // Current active snapshot
  const activeSnapshot = snapshots[currentStep] || snapshots[0];
  const previousSnapshot = currentStep > 0 ? snapshots[currentStep - 1] : undefined;
  const events = snapshots.map((s) => s.event).filter(Boolean) as any[];

  const lastBenchmarkSnapshot = benchmarkSnapshots[benchmarkSnapshots.length - 1] || benchmarkSnapshots[0];
  const lastCustomSnapshot = snapshots[snapshots.length - 1] || snapshots[0];

  return (
    <div id="workstation-root" className="min-h-screen bg-slate-950 text-slate-100 flex flex-col font-sans selection:bg-cyan-900 selection:text-white">
      {/* 1. Header / Topic Selector / Execution Status Bar (#49, #57) */}
      <Header
        topics={SIMULATION_TOPICS}
        activeTopic={activeTopic}
        onSelectTopic={handleSelectTopic}
        activePreset={activePreset}
        onSelectPreset={handleSelectPreset}
        mode={mode}
        status={status}
        currentStep={currentStep}
        totalSteps={snapshots.length}
        activeView={activeView}
        onChangeView={setActiveView}
        onOpenDocs={() => setIsDocsOpen(true)}
        onOpenComparison={() => setIsComparisonOpen(true)}
      />

      {/* 2. Simulation Control Deck (#15, #16) */}
      <ControlDeck
        status={status}
        mode={mode}
        currentStep={currentStep}
        totalSteps={snapshots.length}
        playbackSpeed={playbackSpeed}
        activeEventLabel={activeSnapshot?.event?.label}
        onPlay={handlePlay}
        onPause={handlePause}
        onPrevStep={handlePrevStep}
        onNextStep={handleNextStep}
        onReset={handleReset}
        onResetToDefault={handleResetToDefault}
        onSpeedChange={setPlaybackSpeed}
        onToggleParams={() => setIsParamsOpen(!isParamsOpen)}
        isParamsOpen={isParamsOpen}
      />

      {/* 3. Code / Input Workbench: Default Benchmark vs Edit Space (#4, #5, #7) */}
      <Workbench
        benchmarkCode={benchmarkCode}
        editCode={editCode}
        onChangeEditCode={setEditCode}
        activeLine={activeSnapshot?.activeLine}
        view={activeView}
        validationResult={validationResult}
        onExecuteCustom={handleExecuteCustom}
        onCopyToEditSpace={handleCopyToEditSpace}
        isExecuting={status === 'running'}
      />

      {/* Main Workspace Layout: Visualizer, Parameters, Telemetry, Inspector (#49) */}
      <main className="flex-1 p-3 sm:p-4 max-w-[1720px] mx-auto w-full space-y-4">
        {/* Row A: Main Interactive Visualizer & Dynamic Parameter Panel */}
        <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
          {/* Main Visualizer Router */}
          <div className={isParamsOpen ? 'lg:col-span-8 xl:col-span-9' : 'lg:col-span-12'}>
            <VisualizerRouter
              topic={activeTopic}
              state={activeSnapshot.state}
            />
          </div>

          {/* Collapsible / Dynamic Parameter Panel (#17, #18) */}
          {isParamsOpen && (
            <div className="lg:col-span-4 xl:col-span-3">
              <ParameterPanel
                parameters={activeTopic.parameters}
                values={paramValues}
                onChange={handleParamChange}
                onResetParams={handleResetParams}
              />
            </div>
          )}
        </div>

        {/* Row B: Telemetry Matrix (#25, #26, #27) */}
        <TelemetryMatrix
          metrics={activeTopic.metrics}
          values={activeSnapshot.metrics || {}}
          previousValues={previousSnapshot?.metrics}
          topicId={activeTopic.id}
          step={currentStep}
        />

        {/* Row C: Timeline & Chronological Event Log (#23, #24) */}
        <TimelineEventLog
          events={events}
          currentStep={currentStep}
          totalSteps={snapshots.length}
          onSelectStep={(s) => {
            setCurrentStep(s);
            if (status === 'completed') setStatus('paused');
          }}
        />

        {/* Row D: Deep Technical Inspector (#28, #29) */}
        <DeepInspector
          explanation={activeSnapshot.explanation}
          depth={explanationDepth}
          onChangeDepth={setExplanationDepth}
        />
      </main>

      {/* Before / After Comparison Modal (#30, #31, #72) */}
      <ComparisonModal
        isOpen={isComparisonOpen}
        onClose={() => setIsComparisonOpen(false)}
        metrics={activeTopic.metrics}
        benchmarkMetrics={lastBenchmarkSnapshot?.metrics || {}}
        customMetrics={lastCustomSnapshot?.metrics || {}}
        benchmarkTitle="Default Benchmark (Canonical)"
        customTitle={mode === 'CUSTOM' ? 'Custom Run' : 'Current Active Plan'}
      />

      {/* Architecture & Documentation Guide Modal (#61, #62) */}
      <TopicDocsModal
        isOpen={isDocsOpen}
        onClose={() => setIsDocsOpen(false)}
        topicName={activeTopic.name}
        docs={activeTopic.docs}
      />
    </div>
  );
}

function highlightSqlKeywords(line: string): string {
  const trimmed = line.trim().toUpperCase();
  if (trimmed.startsWith('SELECT') || trimmed.startsWith('FROM') || trimmed.startsWith('WHERE') || trimmed.startsWith('JOIN') || trimmed.startsWith('ORDER BY') || trimmed.startsWith('GROUP BY')) {
    return 'text-cyan-300 font-medium';
  }
  if (line.includes('AND ') || line.includes('OR ') || line.includes('ON ')) {
    return 'text-indigo-300';
  }
  return 'text-slate-200';

export interface CodegenConfig {
  joinType: string;
  hasFilter: boolean;
  filterPushedToScan: boolean;
  columnsPruned: boolean;
  wholeStageEnabled: boolean;
  tableName1: string;
  tableName2?: string;
}

export function generateJaninoJavaCode(config: CodegenConfig): string {
  if (!config.wholeStageEnabled) {
    return `// Whole-Stage Code Generation is DISABLED (spark.sql.codegen.wholeStage = false)
// Spark falls back to Volcano Iterator Model with virtual method invocation:
//
// while (child.hasNext()) {
//     InternalRow row = child.next();
//     if (filter.eval(row)) {
//         return project.eval(row);
//     }
// }
// ⚠️ High CPU cache misses and heavy virtual function call dispatch per row.`;
  }

  const isBroadcast = config.joinType === 'BroadcastHashJoin';

  return `/* 
 * Apache Spark Whole-Stage Java Code Generation (Janino Compiler)
 * Stage 1 / GeneratedIteratorForCodegenStage1
 * Project Tungsten: Zero virtual dispatch, unrolled CPU pipelining
 */
package org.apache.spark.sql.catalyst.expressions;

import org.apache.spark.sql.execution.BufferedRowIterator;
import org.apache.spark.sql.catalyst.InternalRow;
import org.apache.spark.sql.catalyst.expressions.codegen.UnsafeRowWriter;
import org.apache.spark.sql.execution.joins.LongHashedRelation;

public final class GeneratedIteratorForCodegenStage1 extends BufferedRowIterator {
    private scala.collection.Iterator<InternalRow> inputScan;
    ${isBroadcast ? 'private LongHashedRelation broadcastRelation;' : ''}
    private UnsafeRowWriter rowWriter;

    public GeneratedIteratorForCodegenStage1(Object[] references) {
        this.inputScan = (scala.collection.Iterator<InternalRow>) references[0];
        ${isBroadcast ? 'this.broadcastRelation = (LongHashedRelation) references[1];' : ''}
        this.rowWriter = new UnsafeRowWriter(6, 64);
    }

    @Override
    protected void processNext() throws java.io.IOException {
        // Direct unrolled while loop over partition batches (no virtual next() calls)
        while (inputScan.hasNext()) {
            InternalRow scanRow = (InternalRow) inputScan.next();

            ${!config.filterPushedToScan && config.hasFilter ? `// In-Memory Filter evaluation
            long amount = scanRow.getLong(3);
            if (!(amount > 25000L)) {
                continue; // Predicate rejected
            }` : '// [Filter Pushdown Active]: Predicate evaluated in Parquet Vectorized Reader'}

            ${isBroadcast ? `// Fast Broadcast Hash Table Probe (O(1) memory lookup)
            long joinKey = scanRow.getLong(1);
            InternalRow matchedBuildRow = broadcastRelation.getValue(joinKey);
            if (matchedBuildRow == null) {
                continue; // Inner join condition failed
            }` : `// Stream side record passed to downstream SortMergeJoin buffer`}

            // Materialize Tungsten UnsafeRow into off-heap memory buffer
            rowWriter.reset();
            rowWriter.write(0, scanRow.getLong(0));   // order_id
            rowWriter.write(1, scanRow.getLong(1));   // customer_id
            ${isBroadcast ? 'rowWriter.write(2, matchedBuildRow.getUTF8String(1)); // customer_name' : ''}
            
            append(rowWriter.getRow());
            if (shouldStop()) return;
        }
    }
}`;
}

export interface ParsedQuery {
  raw: string;
  selectColumns: Array<{ name: string; alias?: string; line: number }>;
  fromTable: { name: string; alias?: string; line: number };
  joins: Array<{
    type: 'INNER' | 'LEFT' | 'RIGHT' | 'FULL';
    table: string;
    alias?: string;
    condition: string;
    line: number;
  }>;
  whereClauses: Array<{ expression: string; line: number }>;
  groupByColumns: Array<{ name: string; line: number }>;
  havingClauses: Array<{ expression: string; line: number }>;
  orderByColumns: Array<{ name: string; direction: 'ASC' | 'DESC'; line: number }>;
  lineMap: {
    selectLine: number;
    fromLine: number;
    whereLine: number;
    joinLines: number[];
    groupByLine?: number;
    orderByLine?: number;
  };

export function parseSparkQuery(queryText: string): { parsed?: ParsedQuery; error?: string } {
  const lines = queryText.split('\n');
  const cleanLines = lines.map((l) => l.trim());

  if (!queryText.trim()) {
    return { error: 'Empty query. Please provide a valid Spark SQL query or DataFrame operation.' };
  }

  // Find key clauses
  let selectStart = -1;
  let fromStart = -1;
  let whereStart = -1;
  let groupByStart = -1;
  let havingStart = -1;
  let orderByStart = -1;

  for (let i = 0; i < cleanLines.length; i++) {
    const line = cleanLines[i].toUpperCase();
    if (line.startsWith('SELECT')) selectStart = i + 1;
    else if (line.startsWith('FROM')) fromStart = i + 1;
    else if (line.startsWith('WHERE')) whereStart = i + 1;
    else if (line.startsWith('GROUP BY')) groupByStart = i + 1;
    else if (line.startsWith('HAVING')) havingStart = i + 1;
    else if (line.startsWith('ORDER BY')) orderByStart = i + 1;
  }

  if (selectStart === -1) {
    return { error: 'Syntax Error: Missing SELECT statement in query.' };
  }
  if (fromStart === -1) {
    return { error: 'Syntax Error: Missing FROM clause in query.' };
  }

  // Extract from table
  const fromLineIdx = fromStart - 1;
  const fromLineText = lines[fromLineIdx].replace(/--.*$/, '').trim();
  const fromMatch = fromLineText.match(/FROM\s+([a-zA-Z0-9_]+)(?:\s+([a-zA-Z0-9_]+))?/i);
  if (!fromMatch) {
    return { error: `Invalid FROM clause at line ${fromStart}: Expected table identifier.` };
  }

  const fromTable = {
    name: fromMatch[1].toLowerCase(),
    alias: fromMatch[2] ? fromMatch[2].toLowerCase() : fromMatch[1].toLowerCase(),
    line: fromStart,
  };

  // Extract joins
  const joins: ParsedQuery['joins'] = [];
  const joinLines: number[] = [];
  for (let i = 0; i < lines.length; i++) {
    const rawLine = lines[i].replace(/--.*$/, '').trim();
    const joinMatch = rawLine.match(/(?:(INNER|LEFT|RIGHT|FULL)\s+)?JOIN\s+([a-zA-Z0-9_]+)(?:\s+([a-zA-Z0-9_]+))?\s+ON\s+(.*)/i);
    if (joinMatch) {
      const type = (joinMatch[1]?.toUpperCase() || 'INNER') as any;
      joins.push({
        type,
        table: joinMatch[2].toLowerCase(),
        alias: joinMatch[3] ? joinMatch[3].toLowerCase() : joinMatch[2].toLowerCase(),
        condition: joinMatch[4].trim(),
        line: i + 1,
      });
      joinLines.push(i + 1);
    }
  }

  // Extract where clauses
  const whereClauses: ParsedQuery['whereClauses'] = [];
  if (whereStart !== -1) {
    let i = whereStart - 1;
    let whereText = lines[i].replace(/--.*$/, '').replace(/WHERE/i, '').trim();
    if (whereText) {
      whereClauses.push({ expression: whereText, line: i + 1 });
    }
    i++;
    while (i < lines.length) {
      const l = lines[i].replace(/--.*$/, '').trim();
      if (!l) {
        i++;
        continue;
      }
      const upper = l.toUpperCase();
      if (upper.startsWith('GROUP BY') || upper.startsWith('ORDER BY') || upper.startsWith('HAVING') || upper.startsWith('LIMIT')) {
        break;
      }
      if (upper.startsWith('AND ') || upper.startsWith('OR ')) {
        whereClauses.push({ expression: l.replace(/^(AND|OR)\s+/i, '').trim(), line: i + 1 });
      } else if (l.includes('=')) {
        whereClauses.push({ expression: l, line: i + 1 });
      }
      i++;
    }
  }

  // Extract select columns
  const selectColumns: ParsedQuery['selectColumns'] = [];
  let curLine = selectStart;
  while (curLine <= lines.length && curLine !== fromStart) {
    const l = lines[curLine - 1].replace(/--.*$/, '').replace(/SELECT/i, '').trim();
    if (l) {
      const parts = l.split(',');
      for (const p of parts) {
        const trimmed = p.trim();
        if (trimmed) {
          const aliasMatch = trimmed.match(/(.+)\s+AS\s+([a-zA-Z0-9_]+)/i);
          if (aliasMatch) {
            selectColumns.push({ name: aliasMatch[1].trim(), alias: aliasMatch[2].trim(), line: curLine });
          } else {
            selectColumns.push({ name: trimmed, line: curLine });
          }
        }
      }
    }
    curLine++;
  }

  return {
    parsed: {
      raw: queryText,
      selectColumns,
      fromTable,
      joins,
      whereClauses,
      groupByColumns: [],
      havingClauses: [],
      orderByColumns: [],
      lineMap: {
        selectLine: selectStart,
        fromLine: fromStart,
        whereLine: whereStart !== -1 ? whereStart : fromStart,
        joinLines,
        groupByLine: groupByStart !== -1 ? groupByStart : undefined,
        orderByLine: orderByStart !== -1 ? orderByStart : undefined,
      },
    },
  };
}

export function validateSparkQuery(queryText: string, validTables: string[]): ValidationResult {
  const { parsed, error } = parseSparkQuery(queryText);
  if (error || !parsed) {
    return {
      isValid: false,
      issues: [{ message: error || 'Failed to parse query', severity: 'error' }],
    };
  }

  const issues: ValidationResult['issues'] = [];

  // Validate primary table exists in catalog
  if (!validTables.includes(parsed.fromTable.name)) {
    issues.push({
      line: parsed.fromTable.line,
      message: `Table '${parsed.fromTable.name}' not found in Spark Catalog. Available tables: ${validTables.join(', ')}`,
      severity: 'error',
    });
  }

  // Validate joins
  for (const j of parsed.joins) {
    if (!validTables.includes(j.table)) {
      issues.push({
        line: j.line,
        message: `Joined table '${j.table}' not found in Spark Catalog. Available tables: ${validTables.join(', ')}`,
        severity: 'error',
      });
    }
    if (!j.condition) {
      issues.push({
        line: j.line,
        message: `Join condition missing for table '${j.table}'. Cross joins require explicit Cartesian product syntax.`,
        severity: 'warning',
      });
    }
  }

  return {
    isValid: issues.filter((i) => i.severity === 'error').length === 0,
    issues,
  };
}
